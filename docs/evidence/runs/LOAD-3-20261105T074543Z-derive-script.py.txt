def entries(win, name):
    for e in W["entries"]:
        if e.get("window")==win and e.get("name")==name: return e.get("result",[])
    return []
def table(win, name, keys=("namespace","container")):
    out={}
    for r in entries(win,name):
        out[tuple(r["metric"].get(k,"") for k in keys)]=float(r["value"][1])
    return out

def series(win,name):
    for e in SER["entries"]:
        if e.get("window")==win and e.get("name")==name: return e.get("result",[])
    return []
PLANE11={"plane-web","plane-admin","plane-space","plane-live","plane-api","plane-worker","plane-beat-worker","plane-pgdb","plane-redis","plane-rabbitmq","plane-silo"}
def group_of(m):
    ns,pod,c=m.get("namespace",""),m.get("pod",""),m.get("container","")
    if ns in ("plane-dev","plane-prod-sim"):
        if c in ("postgres-exporter","redis-exporter") or pod.startswith(("postgres-exporter","redis-exporter")): return "exporters"
        if pod.startswith("iso-"): return "infisical"
        if pod.startswith("k6-"): return "k6"
        if re.match(r"plane-(backup|restore|api-migrate|rollback)|.*-migrate-|.*-(guard)", pod): return f"jobs-{ns}"
        if c in PLANE11 or re.match(r"plane-(web|admin|space|live|api|worker|beat-worker|pgdb|redis|rabbitmq|minio|silo)-wl", pod): return f"plane-{ns}"
        return f"other-{ns}"
    if ns=="traefik": return "traefik+metrics-server"
    if ns=="kube-system":
        return "traefik+metrics-server" if pod.startswith("metrics-server") else "control-plane"
    if ns=="local-path-storage": return "control-plane"
    if ns=="argocd": return "argocd"
    if ns=="infisical": return "infisical"
    if ns in("monitoring","monitoring-agents"):
        if pod.startswith(("loki","alloy")) or c in("loki","alloy"): return "loki+alloy"
        return "monitoring"
    return "other-"+ns
def group_table(win):
    mem,cpu={}, {}
    pods={}
    for name,d in(("mem",mem),("cpu",cpu)):
        for s in series(win,name):
            g=group_of(s["metric"])
            pods.setdefault(g,set()).add(s["metric"].get("namespace","")+"/"+s["metric"].get("pod","")+"/"+s["metric"].get("container",""))
            for t,v in s["values"]:
                d.setdefault(g,{}).setdefault(t,0.0); d[g][t]+=float(v)
    ret={}
    for g in sorted(set(mem)|set(cpu)):
        mv=list(mem.get(g,{}).values()); cv=list(cpu.get(g,{}).values())
        pm=max(mem.get(g,{1:0}).items(),key=lambda x:x[1]) if mv else (0,0)
        pc=max(cpu.get(g,{1:0}).items(),key=lambda x:x[1]) if cv else (0,0)
        ret[g]=dict(mem_avg=sum(mv)/len(mv)/MIB if mv else 0,mem_max=pm[1]/MIB,mem_at=pm[0],cpu_avg=sum(cv)/len(cv)*1000 if cv else 0,cpu_max=pc[1]*1000,cpu_at=pc[0],n=len(pods[g]))
    return ret,pods

def p95(bs):
    bs=[(float(b["le"]) if b["le"]!="+Inf" else float("inf"),b["count"]) for b in bs]
    bs.sort()
    tot=bs[-1][1]; rank=0.95*tot; prev_le,prev_c=0.0,0.0
    for le,c in bs:
        if c>=rank:
            return tot,rank,prev_le,le,prev_le+(rank-prev_c)/(c-prev_c)*(le-prev_le)
        prev_le,prev_c=le,c
for e in B["entries"]:
    tot,rank,lo,hi,v=p95(e["buckets"])
    print(e["window"],e["method"],f"count {tot:.0f} p95 rank {rank:.1f} bucket {lo}-{hi} s interpolated {v*1000:.0f} ms")
