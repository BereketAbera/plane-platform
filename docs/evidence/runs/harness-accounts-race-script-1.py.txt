#!/usr/bin/env python3
"""Two smoke runs, one for each environment, that both read the accounts file before either writes it. On a temp file.

Accounts as it was before the lock came in, from Git at the commit before 0bba707, and as it is in the working tree. Fake values
only, and the real credentials file is not opened.

usage: accounts_race.py <repository root>
"""
import importlib.util
import subprocess
import sys
import tempfile
from pathlib import Path

root = Path(sys.argv[1])


def load(name: str, path: Path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def race(module) -> str:
    with tempfile.TemporaryDirectory() as directory:
        path = Path(directory) / "accounts.env"
        dev, prod_sim = module.Accounts(path, "dev"), module.Accounts(path, "prod-sim")
        dev.set(API_TOKEN="token-of-the-dev-run")
        prod_sim.set(API_TOKEN="token-of-the-prod-sim-run")
        kept_dev = module.Accounts(path, "dev").get("API_TOKEN") == "token-of-the-dev-run"
        kept_prod_sim = module.Accounts(path, "prod-sim").get("API_TOKEN") == "token-of-the-prod-sim-run"
        return f"dev token kept: {kept_dev} prod-sim token kept: {kept_prod_sim}"


with tempfile.TemporaryDirectory() as directory:
    # parents[2] of the file is what smoke.py takes for its root, so give it that depth.
    old_file = Path(directory) / "a" / "b" / "smoke_before.py"
    old_file.parent.mkdir(parents=True)
    old_file.write_text(subprocess.run(["git", "-C", str(root), "show", "0bba707~1:tests/e2e/smoke.py"], capture_output=True, text=True, check=True).stdout)
    print("before the fix", race(load("smoke_before", old_file)))
print("after the fix", race(load("smoke_after", root / "tests" / "e2e" / "smoke.py")))
