--- cg-pod.sh
#!/bin/bash
# Reads memory cgroup files on the node for a pod's cgroup and each of its containers' cgroups. Read only.
export KUBECONFIG=/Users/bek/dev/pr/plane-platform/.local/kubeconfig
KC=/Users/bek/dev/pr/plane-platform/.tools/bin/kubectl
NODE=plane-platform-control-plane
for spec in "$@"; do
  ns=${spec%/*}; pod=${spec#*/}
  uid=$("$KC" --context kind-plane-platform -n "$ns" get pod "$pod" -o jsonpath='{.metadata.uid}')
  u=$(printf '%s' "$uid" | tr '-' '_')
  qos=$("$KC" --context kind-plane-platform -n "$ns" get pod "$pod" -o jsonpath='{.status.qosClass}' | tr 'A-Z' 'a-z')
  dir="/sys/fs/cgroup/kubelet.slice/kubelet-kubepods.slice/kubelet-kubepods-${qos}.slice/kubelet-kubepods-${qos}-pod${u}.slice"
  echo "### $spec uid=$uid qos=$qos"
  "$KC" --context kind-plane-platform -n "$ns" get pod "$pod" -o jsonpath='{range .status.containerStatuses[*]}{"status: "}{.name}{" id="}{.containerID}{" restarts="}{.restartCount}{" started="}{.state.running.startedAt}{" lastExit="}{.lastState.terminated.exitCode}{" "}{.lastState.terminated.reason}{" lastFinished="}{.lastState.terminated.finishedAt}{"\n"}{end}'
  "$KC" --context kind-plane-platform -n "$ns" get pod "$pod" -o jsonpath='{range .spec.containers[*]}{"limit: "}{.name}{" "}{.resources.limits.memory}{"\n"}{end}{range .spec.initContainers[*]}{"init limit: "}{.name}{" "}{.resources.limits.memory}{"\n"}{end}'
  docker exec "$NODE" sh -c '
    d="$1"
    show() { # label dir
      lbl="$1"; c="$2"
      printf -- "-- %s\n" "$lbl"
      printf "memory.max %s\n" "$(cat "$c/memory.max")"
      printf "memory.peak %s\n" "$(cat "$c/memory.peak" 2>/dev/null)"
      printf "memory.current %s\n" "$(cat "$c/memory.current")"
      printf "memory.events:"; while read -r k v; do printf " %s=%s" "$k" "$v"; done <"$c/memory.events"; printf "\n"
      printf "memory.stat:"; grep -E "^(anon|file|kernel|slab|shmem|pgscan|pgsteal|workingset_refault_file) " "$c/memory.stat" | while read -r k v; do printf " %s=%s" "$k" "$v"; done; printf "\n"
    }
    show "pod cgroup $(basename "$d")" "$d"
    for c in "$d"/*/; do
      [ -f "$c/memory.max" ] || continue
      show "container cgroup $(basename "$c")" "$c"
    done
  ' sh "$dir"
done
--- cg-all.sh
#!/bin/bash
# One line per pod, read only, on the node: the pod cgroup's and each limited container's memory.events max, peak and anon.
export KUBECONFIG=/Users/bek/dev/pr/plane-platform/.local/kubeconfig
KC=/Users/bek/dev/pr/plane-platform/.tools/bin/kubectl
NODE=plane-platform-control-plane
printf 'namespace/pod | pod memory.max MiB | pod max (local) | pod oom_kill | pod peak MiB | pod anon MiB | container: limit MiB, max, peak MiB ...\n'
for ns in "$@"; do
  "$KC" --context kind-plane-platform -n "$ns" get pods --field-selector=status.phase=Running -o jsonpath='{range .items[*]}{.metadata.name}{" "}{.metadata.uid}{" "}{.status.qosClass}{"\n"}{end}' |
  while read -r pod uid qos; do
    u=$(printf '%s' "$uid" | tr '-' '_'); q=$(printf '%s' "$qos" | tr 'A-Z' 'a-z')
    if [ "$q" = guaranteed ]; then dir="/sys/fs/cgroup/kubelet.slice/kubelet-kubepods.slice/kubelet-kubepods-pod${u}.slice"; else dir="/sys/fs/cgroup/kubelet.slice/kubelet-kubepods.slice/kubelet-kubepods-${q}.slice/kubelet-kubepods-${q}-pod${u}.slice"; fi
    docker exec "$NODE" sh -c '
      d="$1"; pp="$2"
      mi() { [ "$1" = max ] && { printf max; return; }; awk -v v="$1" "BEGIN{printf \"%.0f\", v/1048576}"; }
      ev() { awk -v k="$2" "\$1==k{print \$2}" "$1"; }
      st() { awk -v k="$2" "\$1==k{print \$2}" "$1"; }
      printf "%s | %s | %s (%s) | %s | %s | %s |" "$pp" "$(mi "$(cat "$d/memory.max")")" "$(ev "$d/memory.events" max)" "$(ev "$d/memory.events.local" max)" "$(ev "$d/memory.events" oom_kill)" "$(mi "$(cat "$d/memory.peak")")" "$(mi "$(st "$d/memory.stat" anon)")"
      for c in "$d"/*/; do
        [ -f "$c/memory.max" ] || continue
        m="$(cat "$c/memory.max")"; [ "$m" = max ] && continue
        printf " %s,%s,%s" "$(mi "$m")" "$(ev "$c/memory.events" max)" "$(mi "$(cat "$c/memory.peak")")"
      done
      printf "\n"
    ' sh "$dir" "$ns/$pod"
  done
done
