#!/usr/bin/env python3
"""CI-4 and CI-5: reads the run log's walk of what GHCR served, the pull request's digests, and the local registry, and compares them.
Reads only. Prints everything it compares, so the output is the evidence."""
import json
import re
import subprocess
import sys
import urllib.request

LOG = "/Users/bek/dev/pr/plane-platform/docs/evidence/ci/ci-4-20261105T053955Z-run-41975430939.log"
ACCEPT = "application/vnd.oci.image.manifest.v1+json, application/vnd.docker.distribution.manifest.v2+json, application/vnd.oci.image.index.v1+json"
LOCAL = {
    "web": "sha256:3d7575a1b9341b24dc882356b24d3d45126a4f5ccfb6abc72b07699668543747",
    "admin": "sha256:ce7adc37b1a7ff2f9f04457f89f7758152c9740be8e7f801b3ff0ae2a7a9c796",
    "space": "sha256:5f0119ff2266a7f0493e5903f1fdf47c022cc5ca86b6f8fa14d8485513cc0039",
    "backend": "sha256:8c04bd18bb0823f5740881cbed2f8759a98729a9c687b4009f0a3641c853adbe",
}
ANSI = re.compile(r"\x1b\[[0-9;]*m|\^\[\[[0-9;]*m")


def lines():
    with open(LOG, encoding="utf-8", errors="replace") as handle:
        for raw in handle:
            parts = raw.rstrip("\n").split("\t")
            if len(parts) >= 3:
                text = "\t".join(parts[2:])
                text = re.sub(r"^﻿?\d{4}-\d\d-\d\dT[\d:.]+Z ", "", text)
                yield parts[0], parts[1], ANSI.sub("", text)


walks = {}  # (registry host, name) -> {digest, config, layers:[(short, mib, entries, state)], summary}
current = None
published = {}
recorded_output = []
scan = []
for job, step, text in lines():
    if job != "Build and push":
        if job == "Open digest bump for dev" and "DIGESTS: " in text:
            recorded_output.append(text)
        continue
    if step.startswith("Scan what"):
        scan.append(text)
        continue
    match = re.match(r"== ((localhost:5059|ghcr\.io)/\S*?/([a-z]+))@(sha256:[0-9a-f]{64})", text)
    if match:
        current = (match.group(2), match.group(3))
        walks[current] = {"digest": match.group(4), "layers": [], "summary": None, "config": None}
        continue
    if current:
        layer = re.match(r"layer\s+(\d+)/(\d+) (sha256:[0-9a-f]{12})\s+([\d.]+) MiB\s+(\d+) entries, (\d+) archives opened: (.+)$", text)
        if layer:
            walks[current]["layers"].append((layer.group(3), float(layer.group(4)), int(layer.group(5)), layer.group(7)))
            continue
        config = re.match(r"config (sha256:[0-9a-f]{64}): (\w+)", text)
        if config:
            walks[current]["config"] = (config.group(1), config.group(2))
            continue
        summary = re.match(r"(clean|HIT|hit|incomplete)\S*\s+(\d+) hit\(s\)\s+(\d+) not opened (\S+)", text)
        if summary:
            walks[current]["summary"] = text.strip()
            continue
    ok = re.match(r"\s*ok (ghcr\.io/\S+/([a-z]+))@(sha256:[0-9a-f]{64})$", text)
    if ok:
        published[ok.group(2)] = ok.group(3)

print("=== 1. What the run's second walk read from ghcr.io, per image, compared with its first walk in the throwaway registry")
total_all = 0.0
unique = {}
for name in ("web", "admin", "space", "backend"):
    ghcr = walks[("ghcr.io", name)]
    staging = walks[("localhost:5059", name)]
    same = [l[0] for l in ghcr["layers"]] == [l[0] for l in staging["layers"]] and ghcr["digest"] == staging["digest"] and ghcr["config"] == staging["config"]
    size = sum(l[1] for l in ghcr["layers"])
    total_all += size
    for layer in ghcr["layers"]:
        unique[layer[0]] = layer[1]
    print(f"{name}: digest {ghcr['digest']}")
    print(f"  recorded as published: {published.get(name)}, same as walked: {published.get(name) == ghcr['digest']}")
    print(f"  config {ghcr['config'][0]} {ghcr['config'][1]}")
    print(f"  {len(ghcr['layers'])} layers, {sum(1 for l in ghcr['layers'] if l[3] == 'clean')} clean, {sum(l[2] for l in ghcr['layers'])} tar entries, compressed {size:.1f} MiB (sum of the rounded values in the log)")
    print(f"  summary line on ghcr.io:   {ghcr['summary']}")
    print(f"  summary line in staging:   {staging['summary']}")
    print(f"  same digest, config and layer list in both walks: {same}")
print(f"all four, counting a shared layer once per image: {total_all:.1f} MiB. Counting each distinct layer once: {sum(unique.values()):.1f} MiB")

print()
print("=== 2. The dev digest bump pull request, read now, against what the run recorded")
diff = subprocess.run(["gh", "pr", "diff", "1", "--repo", "BereketAbera/plane-platform"], capture_output=True, text=True, check=True).stdout
proposed = {}
for name, digest in re.findall(r"^\+.*localhost:5001/wayline/([a-z]+)@(sha256:[0-9a-f]{64})", diff, re.M):
    proposed.setdefault(name, set()).add(digest)
overlay_lines = re.findall(r"^\+- digest: (sha256:[0-9a-f]{64})", diff, re.M)
for name in ("web", "admin", "space", "backend"):
    print(f"{name}: proposed {sorted(proposed.get(name, []))}, the run published {published.get(name)}, match: {proposed.get(name) == {published.get(name)}}")
print(f"digests the overlay diff adds: {len(overlay_lines)}, all of them published by the run: {set(overlay_lines) == set(published.values())}")

print()
print("=== 3. Those digests against what the two environments run now, read from the local registry")
def local_manifest(name, digest):
    request = urllib.request.Request(f"http://localhost:5001/v2/wayline/{name}/manifests/{digest}", headers={"Accept": ACCEPT})
    with urllib.request.urlopen(request, timeout=20) as reply:
        return reply.headers.get("Docker-Content-Digest"), json.load(reply)

def local_has(name, digest):
    request = urllib.request.Request(f"http://localhost:5001/v2/wayline/{name}/manifests/{digest}", headers={"Accept": ACCEPT}, method="HEAD")
    try:
        with urllib.request.urlopen(request, timeout=20) as reply:
            return reply.status
    except urllib.error.HTTPError as err:
        return err.code

for name in ("web", "admin", "space", "backend"):
    served, manifest = local_manifest(name, LOCAL[name])
    local_layers = [layer["digest"][:19] for layer in manifest["layers"]]
    ghcr_layers = [layer[0] for layer in walks[("ghcr.io", name)]["layers"]]
    shared = [digest for digest in ghcr_layers if digest in local_layers]
    only_ci = [digest for digest in ghcr_layers if digest not in local_layers]
    only_local = [digest for digest in local_layers if digest not in ghcr_layers]
    sizes = {layer["digest"][:19]: layer["size"] for layer in manifest["layers"]}
    ci_sizes = {layer[0]: layer[1] for layer in walks[("ghcr.io", name)]["layers"]}
    print(f"{name}: local {served}, ghcr {published.get(name)}, same: {served == published.get(name)}")
    print(f"  layers: local {len(local_layers)}, ghcr {len(ghcr_layers)}, in both {len(shared)}")
    print(f"  only in the ghcr image: {[(digest, ci_sizes[digest]) for digest in only_ci]}")
    print(f"  only in the local image: {[(digest, round(sizes[digest] / 1048576, 1)) for digest in only_local]}")
    print(f"  the local registry holds the ghcr digest now: HTTP {local_has(name, published[name])}")

print()
print("=== 4. The scan step of the run, as logged")
for text in scan:
    if text.startswith("##[") or text.startswith("shell:") or text.startswith("env:") or text.startswith("  "):
        continue
    print(text)
