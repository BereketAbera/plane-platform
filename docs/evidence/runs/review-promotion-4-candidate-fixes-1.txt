#!/usr/bin/env python3
# Root-cause candidate for which dump to name: record each publish, and read when the target first went out.
import sys
root = sys.argv[1]
def sub(path, old, new):
    p = root + '/' + path
    s = open(p).read()
    assert s.count(old) == 1, (path, old[:50], s.count(old))
    open(p, 'w').write(s.replace(old, new))
sub('scripts/gitops-publish.sh',
    '''ok "published main ${new:0:12} to $BARE"
''',
    '''ok "published main ${new:0:12} to $BARE"
mkdir -p "$LOCAL_DIR/evidence-raw"
printf '%s %s\\n' "$(date -u +%Y%m%dT%H%M%SZ)" "$new" >>"$LOCAL_DIR/evidence-raw/publishes.log"
''')
sub('scripts/rollback.sh',
    '''newest_dump() { # newest_dump [<commit>]: the newest finished backup, from no later than that commit
  local dir name latest="99999999T999999Z" newest=""
  if [[ $# -gt 0 ]]; then latest="$(TZ=UTC git log -1 --date='format-local:%Y%m%dT%H%M%SZ' --format=%cd "$1")"; fi
''',
    '''published_at() { # published_at <commit>: when the first publish holding it went out, if that was recorded
  local at sha log="$LOCAL_DIR/evidence-raw/publishes.log"
  [[ -f "$log" ]] || return 0
  while read -r at sha; do
    if git merge-base --is-ancestor "$1" "$sha" 2>/dev/null; then printf '%s' "$at"; return 0; fi
  done <"$log"
}
newest_dump() { # newest_dump [<commit>]: the newest finished backup, from no later than that commit reached Argo CD
  local dir name latest="99999999T999999Z" newest=""
  if [[ $# -gt 0 ]]; then
    latest="$(published_at "$1")"
    [[ -n "$latest" ]] || latest="$(TZ=UTC git log -1 --date='format-local:%Y%m%dT%H%M%SZ' --format=%cd "$1")"
  fi
''')
