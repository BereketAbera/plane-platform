BY_ID = {d[0]: d for d in DEFECTS}


def plant(root, ids):
    for one in ids:
        _, name, _, path, old, new = BY_ID[one]
        full = root + "/" + path
        text = open(full).read()
        count = text.count(old)
        if count != 1:
            print(f"COULD NOT PLANT {one}: {name}: {count} matches")
            sys.exit(3)
        open(full, "w").write(text.replace(old, new))


def main():
    root, verb, *rest = sys.argv[1:]
    if verb == "list":
        for one in DEFECTS:
            print(f"{one[0]}\t{one[2]}\t{one[1]}")
    elif verb == "plant":
        plant(root, rest)
    else:
        sys.exit(f"unknown verb {verb}")

#!/bin/bash
# plant-one.sh <id> <scenario>: one defect, alone, in a scratch clone of main's HEAD, and the one scenario meant to catch it.
# A line of output per defect: CAUGHT or MISSED, the flows' exit status, and what the scenario expected.
S=/private/tmp/claude-501/-Users-bek-dev-pr/58f1388d-0d93-4cd1-966b-9c80982ad690/scratchpad/ci-track
MAIN=/Users/bek/dev/pr/plane-platform
id="$1"; scenario="$2"
W="$S/plant-work/$id"
rm -rf "${W:?}"; mkdir -p "$W"
git clone --quiet "$MAIN" "$W/c" || { echo "$id SETUP-FAILED"; exit 0; }
git -C "$W/c" checkout --quiet "$PLANT_COMMIT" || { echo "$id SETUP-FAILED"; exit 0; }
mkdir -p "$W/c/.tools"; ln -s "$MAIN/.tools/bin" "$W/c/.tools/bin"
python3 -B "$S/plant.py" "$W/c" plant "$id" >"$W/plant.out" 2>&1 || { echo "$id COULD-NOT-PLANT $(cat "$W/plant.out")"; rm -rf "${W:?}"; exit 0; }
(cd "$W/c" && ./scripts/ci/test-promotion-flows.sh "$scenario") >"$W/flows.out" 2>&1
rc=$?
expected="$(sed 's/\x1b\[[0-9;]*m//g' "$W/flows.out" | sed -n 's/^warn   expected: //p;s/^warn   a command in the scenario itself failed.*/(a command in the scenario stopped it)/p' | head -n 3 | tr '\n' ';')"
ran="$(sed 's/\x1b\[[0-9;]*m//g' "$W/flows.out" | grep -cE "^( ok|warn) $scenario( FAILED)?")"
if [[ "$rc" -ne 0 && "$ran" -ge 1 ]]; then verdict=CAUGHT; else verdict=MISSED; fi
printf '%s\t%s\texit %s\t%s\t%s\n' "$id" "$verdict" "$rc" "$scenario" "$expected"
if [[ "$verdict" == MISSED ]]; then sed 's/\x1b\[[0-9;]*m//g' "$W/flows.out" | tail -n 8 | cut -c1-200 >&2; fi
rm -rf "${W:?}"
