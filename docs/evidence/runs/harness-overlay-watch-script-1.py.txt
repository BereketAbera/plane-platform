#!/usr/bin/env python3
"""Item 5 on a real rollout: branding-overlay for prod-sim, read every 5s while the promotion rolls out. Only reads."""
import contextlib, io, sys, tempfile, time, datetime as dt
sys.path.insert(0, "/Users/bek/dev/pr/plane-platform/tests/e2e")
import smoke as S
last = None
end = time.monotonic() + 600
passes = 0
while time.monotonic() < end and passes < 3:
    with tempfile.TemporaryDirectory() as d:
        s = S.Smoke(S.parse_args(["--env", "prod-sim", "--accounts", f"{d}/a.env", "--sentinel", f"{d}/s.json"]))
        with contextlib.redirect_stdout(io.StringIO()) as out:
            s.run("branding-overlay", s.step_branding_overlay)
    line = out.getvalue().strip()
    passes = passes + 1 if line.startswith("PASS") else 0
    if line != last:
        print(dt.datetime.now(dt.timezone.utc).strftime("%H:%M:%SZ"), line[:420], flush=True)
        last = line
    time.sleep(5)
print("done", dt.datetime.now(dt.timezone.utc).strftime("%H:%M:%SZ"), flush=True)
