#!/usr/bin/env python3
# LOAD-3: df -k and du -sk per PVC mount, from inside each pod that has one. Read-only. Usage: capture-pvc.py <evidence-dir> <stamp>
import json, subprocess, sys, datetime, os
from concurrent.futures import ThreadPoolExecutor
EV, ST = sys.argv[1:3]
env = dict(os.environ, KUBECONFIG="/Users/bek/dev/pr/plane-platform/.local/kubeconfig")
K = "/Users/bek/dev/pr/plane-platform/.tools/bin/kubectl"
now = lambda: datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
pods = json.loads(subprocess.run([K, "get", "pods", "-A", "-o", "json"], env=env, capture_output=True, text=True, timeout=60).stdout)["items"]
jobs = []
for p in pods:
    if p["status"].get("phase") != "Running": continue
    claims = {v["name"]: v["persistentVolumeClaim"]["claimName"] for v in p["spec"].get("volumes", []) if "persistentVolumeClaim" in v}
    done = set()
    for c in p["spec"]["containers"]:
        for m in c.get("volumeMounts", []):
            if m["name"] in claims and m["name"] not in done:
                done.add(m["name"]); jobs.append((p["metadata"]["namespace"], p["metadata"]["name"], c["name"], claims[m["name"]], m["mountPath"]))
def one(j):
    ns, pod, c, pvc, mnt = j
    head = f"--- {ns}/{pod} container={c} pvc={pvc} mount={mnt} at {now()}"
    out = []
    for cmd in (["df", "-k", mnt], ["du", "-sk", mnt]):
        r = subprocess.run([K, "exec", "-n", ns, pod, "-c", c, "--", *cmd], env=env, capture_output=True, text=True, timeout=40)
        out.append(f"$ {' '.join(cmd)}\n{r.stdout}{r.stderr}" + (f"[could not read: exit {r.returncode}]\n" if r.returncode else ""))
    return head + "\n" + "".join(out)
with ThreadPoolExecutor(4) as ex: res = list(ex.map(one, sorted(jobs)))
with open(f"{EV}/LOAD-3-{ST}-pvc-df.txt", "w") as f:
    f.write(f"# {now()} command: kubectl exec <pod> -c <container> -- df -k <mount>, then du -sk <mount>, for each running pod with a PVC. Node volumes share one filesystem, so df shows that filesystem and du shows the volume.\n")
    f.write("\n".join(res) + "\n")
    f.write(f"# {now()} done\n")

#!/usr/bin/env python3
# LOAD-3: per-minute summary of plane-api's request log, to check what reached the API in each window. Paths have ids
# masked and user ids dropped. Usage: python3 -I access-summary.py <stamp> <out> <dev-log> <prod-sim-log>
import json, sys, re, collections, datetime
ST, OUT, DEV, PROD = sys.argv[1:5]
now = "2026-11-05T09:30:08Z"  # when the logs were read; this script only summarizes the saved copies
WIN = [("idle", "08:47:35", "09:02:35"), ("realistic", "09:02:59", "09:12:02"), ("saturation", "09:12:18", "09:21:21")]
STEPS = [("realistic/vus_05", "09:03:01", "09:06:01"), ("realistic/vus_10", "09:06:01", "09:09:01"), ("realistic/vus_20", "09:09:01", "09:12:01"), ("saturation/vus_05", "09:12:20", "09:15:20"), ("saturation/vus_10", "09:15:20", "09:18:20"), ("saturation/vus_20", "09:18:20", "09:21:20")]
def rows(fn):
    for line in open(fn, errors="replace"):
        ts, _, rest = line.partition(" ")
        if not rest.startswith("{"): continue
        try: d = json.loads(rest)
        except Exception: continue
        if d.get("name") != "plane.api.request": continue
        m = re.match(r"(\w+) (\S+) (\d+)", d.get("message", ""))
        if not m: continue
        p = re.sub(r"[0-9a-f]{8}-[0-9a-f-]{27}", "<id>", m.group(2))
        yield ts[11:19], m.group(1), p, m.group(3), d.get("user_agent", ""), d.get("remote_addr", ""), bool(d.get("user_id") not in (None, "None"))
def kind(p, ua):
    if "__probe__" in p or p == "/api/instances/": return "probe"
    if "/k6-load" in p or ua.startswith("k6") or ua.startswith("Grafana"): return "k6"
    return "other"
with open(OUT, "w") as f:
    f.write(f"# {now} command: kubectl logs -n <plane-dev|plane-prod-sim> deploy/plane-api-wl --since=60m (dev, read at 09:29:25Z) or --since=70m (prod-sim, read at 09:30:08Z) --timestamps, summarized by access-summary.py. Ids masked, user ids dropped.\n")
    for env, fn in (("plane-dev", DEV), ("plane-prod-sim", PROD)):
        R = list(rows(fn))
        f.write(f"\n== {env}: request log lines {len(R)}, first {R[0][0] if R else '-'} last {R[-1][0] if R else '-'} UTC\n")
        for w, a, b in WIN:
            c = collections.Counter(kind(r[2], r[4]) for r in R if a <= r[0] <= b)
            f.write(f"{w} {a} to {b}: {dict(c)}\n")
        f.write("-- everything that is not a probe and not k6, with its user agent and client, first and last time, count\n")
        g = {}
        for t, me, p, co, ua, ra, signed in R:
            if t < "08:44:00" or kind(p, ua) != "other": continue
            k = (me, p, co, ua, ra, signed); x = g.setdefault(k, [t, t, 0]); x[1] = t; x[2] += 1
        for k, x in sorted(g.items(), key=lambda kv: kv[1][0]):
            f.write(f"{x[0]} {x[1]} n={x[2]} {k[0]} {k[1]} {k[2]} ua={k[3]!r} client={k[4]} signed_in={k[5]}\n")
        br = [r[0] for r in R if r[4].startswith("Mozilla")]
        f.write(f"-- browser user agents in this log: {len(br)} requests, first {br[0] if br else '-'} last {br[-1] if br else '-'}; before 08:44:00 {sum(1 for x in br if x < '08:44:00')}, last of those {max([x for x in br if x < '08:44:00'] or ['-'])}\n")
        f.write("-- not probe, not k6, requests per minute\n")
        m = collections.Counter(t[:5] for t, me, p, co, ua, ra, s in R if t >= "08:44:00" and kind(p, ua) == "other")
        f.write(" ".join(f"{k}={v}" for k, v in sorted(m.items())) + "\n")
        f.write("-- k6 user agents seen, count\n")
        u = collections.Counter(ua for t, me, p, co, ua, ra, s in R if kind(p, ua) == "k6")
        f.write(str(dict(u)) + "\n")
        f.write("-- per load step: k6, other, probe\n")
        for st, a, b in STEPS:
            c = collections.Counter(kind(r[2], r[4]) for r in R if a <= r[0] < b)
            f.write(f"{st} {a} to {b}: {dict(c)}\n")
