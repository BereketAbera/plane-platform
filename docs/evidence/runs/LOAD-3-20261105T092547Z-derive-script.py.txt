def p_of(bs, q=0.95):
    bs = [(float(b["le"]) if b["le"] != "+Inf" else float("inf"), b["count"]) for b in bs]; bs.sort()
    tot = bs[-1][1]; rank = q*tot; pl, pc = 0.0, 0.0
    for le, c in bs:
        if c >= rank: return tot, rank, pl, le, (pl+(rank-pc)/(c-pc)*(le-pl) if le != float("inf") and c > pc else None)
        pl, pc = le, c

def ctr_stats(name, a, b, scale):
    out = {}
    for s in series("idle", name):
        m = s["metric"]; k = (m.get("namespace"), m.get("container"))
        for t, v in s["values"]:
            if ep(a) <= t <= ep(b): out.setdefault(k, {}).setdefault(t, 0.0); out[k][t] += float(v)
    return {k: (sum(d.values())/len(d)*scale, max(d.values())*scale, len(d)) for k, d in out.items() if d}

def sum_stats(name, a, b, scale, flt):
    d = {}
    for s in series("idle", name):
        if not flt(s["metric"]): continue
        for t, v in s["values"]:
            if ep(a) <= t <= ep(b): d[t] = d.get(t, 0.0) + float(v)
    p = max(d.items(), key=lambda x: x[1]) if d else (0, 0)
    return (round(sum(d.values())/max(1, len(d))*scale, 1), round(p[1]*scale, 1), ts(p[0]), len(d))
