# What the old and the new PodRestarting expressions read over the retained history, at 30s, for every container in the rule's namespaces. Read only.
import json, subprocess, time, datetime as dt, calendar
def q(expr, start=None, end=None, step=None):
    args = ["/bin/bash", "./q.sh", expr]
    if start is not None:
        args += [str(start), str(end), str(step)]
    d = json.loads(subprocess.run(args, capture_output=True, text=True, check=True).stdout)
    assert d["status"] == "success", d
    return d["data"]["result"]
def z(t): return dt.datetime.fromtimestamp(t, dt.timezone.utc).strftime("%m-%dT%H:%M:%SZ")
def ts(s): return calendar.timegm(time.strptime(s, "%Y-%m-%dT%H:%M:%SZ"))
now=time.time(); start=ts("2026-11-04T21:00:00Z")
sel='kube_pod_container_status_restarts_total{namespace=~"plane-.*|monitoring|monitoring-agents|argocd|traefik|infisical"}'
old='increase(%s[15m]) > 1.5' % sel
new='max_over_time(%s[15m]) - min_over_time(%s[15m]) >= 2' % (sel, sel)
def runs(expr):
    out={}
    # chunks: query_range caps at 11000 points; 30s over 6h is fine
    for s in q(expr,start,now,30):
        m=s["metric"]; k=(m.get("namespace"),m.get("pod"),m.get("container"))
        t=[float(a) for a,_ in s["values"]]; vals=[float(b) for _,b in s["values"]]
        rs=[]; st=t[0]; pv=t[0]; mx=vals[0]
        for x,v in zip(t[1:],vals[1:]):
            if x-pv>30:
                rs.append((st,pv,mx)); st=x; mx=v
            pv=x; mx=max(mx,v)
        rs.append((st,pv,mx))
        out.setdefault(k,[]).extend(rs)
    return out
o=runs(old); n=runs(new)
print("query_range over %s to %s at 30s. old: increase > 1.5. new: max_over_time - min_over_time >= 2" % (z(start), z(now)))
for label,d in (("old",o),("new",n)):
    print("##",label,"true for:")
    for k in sorted(d):
        for a,b,mx in d[k]:
            print("  %s/%s/%s %s to %s, highest %.2f" % (k[0],k[1],k[2],z(a),z(b),mx))
print("## only in old:", sorted(set(o)-set(n)))
print("## only in new:", sorted(set(n)-set(o)))
# Alloy, the restarts README cites
for s in q(sel.replace('}',',pod=~"alloy.*"}'),start,now,30):
    m=s["metric"]; v=[(float(a),float(b)) for a,b in s["values"]]
    ch=[(z(a),b) for (a,b),(c,d) in zip(v[1:],v) if b!=d]
    print("alloy series", m.get("pod"), m.get("container"), "first", z(v[0][0]), "changes", ch)
