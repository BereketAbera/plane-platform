--- scratch-pods.yaml
# Two scratch pods to prove PodRestarting live, from a real restart counter. Not in Git as a manifest, deleted afterwards.
# Each exits with status 1 after 45s, so kube-state-metrics has seen its count at 0 first, as many times as its limit says, then sleeps.
apiVersion: v1
kind: Pod
metadata:
  name: podrestarting-once
  namespace: monitoring
  labels: { scratch: podrestarting }
spec:
  restartPolicy: Always
  terminationGracePeriodSeconds: 0
  automountServiceAccountToken: false
  securityContext: { runAsNonRoot: true, runAsUser: 65534, runAsGroup: 65534, seccompProfile: { type: RuntimeDefault } }
  containers:
    - name: c
      image: docker.io/library/python:3.13.18-alpine@sha256:7984cf56ceaddebe4fefeb75d64000b9f45d8b77b11a09b24524da6cdc3af0c8
      imagePullPolicy: Never
      command: ["sh", "-c", 'n=$(cat /d/n 2>/dev/null || echo 0); echo $((n+1)) >/d/n; if [ "$n" -lt 1 ]; then sleep 45; exit 1; fi; exec sleep 3600']
      volumeMounts: [{ name: d, mountPath: /d }]
      resources: { requests: { cpu: 1m, memory: 8Mi }, limits: { memory: 16Mi } }
      securityContext: { allowPrivilegeEscalation: false, readOnlyRootFilesystem: true, capabilities: { drop: [ALL] } }
  volumes: [{ name: d, emptyDir: { sizeLimit: 1Mi } }]

--- live-watch.sh
#!/bin/bash
# Samples what Prometheus reads for the two scratch pods. Read only. Stops by itself.
W=/private/tmp/claude-501/-Users-bek-dev-pr/58f1388d-0d93-4cd1-966b-9c80982ad690/scratchpad/alerts-work
cd "$W" || exit 9
sel='kube_pod_container_status_restarts_total{namespace="monitoring",pod=~"podrestarting-.*"}'
val() { /bin/bash ./q.sh "$1" | python3 -I -c 'import json,sys; r=json.load(sys.stdin)["data"]["result"]; print(" ".join("%s=%s" % (x["metric"].get("pod","?").replace("podrestarting-",""), round(float(x["value"][1]),2)) for x in sorted(r,key=lambda y:y["metric"].get("pod",""))) or "-")'; }
alerts() { /bin/bash ./q.sh 'ALERTS{alertname="PodRestarting",pod=~"podrestarting-.*"}' | python3 -I -c 'import json,sys; r=json.load(sys.stdin)["data"]["result"]; print(" ".join("%s:%s" % (x["metric"].get("pod","?").replace("podrestarting-",""), x["metric"].get("alertstate")) for x in r) or "-")'; }
end=$(( $(date +%s) + ${1:-1200} ))
while [ "$(date +%s)" -lt "$end" ]; do
  printf '%s count: %s | new: %s | old increase: %s | ALERTS: %s\n' "$(date -u +%H:%M:%SZ)" "$(val "$sel")" "$(val "max_over_time($sel[15m]) - min_over_time($sel[15m])")" "$(val "increase($sel[15m])")" "$(alerts)"
  /bin/sleep 15
done
