### promq.sh
#!/usr/bin/env bash
# Read-only Prometheus queries through the API server's service proxy. Prints "query => result".
set -euo pipefail
K=/Users/bek/dev/pr/plane-platform/.tools/bin/kubectl
KC=/Users/bek/dev/pr/plane-platform/.local/hosted/plane-prod/kubeconfig
while IFS= read -r q; do
  [[ -z "$q" || "$q" == \#* ]] && continue
  enc=$(python3 -I -c 'import sys,urllib.parse; print(urllib.parse.quote(sys.argv[1], safe=""))' "$q")
  out=$("$K" --kubeconfig "$KC" --context plane-prod --request-timeout=60s get --raw "/api/v1/namespaces/monitoring/services/kps-prometheus:9090/proxy/api/v1/query?query=${enc}" 2>&1 || true)
  res=$(printf '%s' "$out" | python3 -I -c '
import json,sys
try:
    d=json.load(sys.stdin)
except Exception as e:
    print("ERR", sys.stdin.read()[:200]); sys.exit()
if d.get("status")!="success":
    print("ERR", d); sys.exit()
r=d["data"]["result"]
out=[]
for s in r:
    m={k:v for k,v in s["metric"].items() if k not in ("__name__","job","endpoint","service","namespace","pod","container","instance")}
    out.append((m, s["value"][1]))
print(out)
')
  printf '%s\n  => %s\n' "$q" "$res"
done

  enc=$(python3 -I -c 'import sys,urllib.parse; print(urllib.parse.quote(sys.argv[1], safe=""))' "$q")

  printf '%s' "$out" | KEEP="${KEEP:-namespace}" python3 -I -c '
import json,os,sys
keep=os.environ["KEEP"].split(",")
raw=sys.stdin.read()
try:
    d=json.loads(raw)
except Exception:
    print("  ERR", raw[:300]); sys.exit()
if d.get("status")!="success":
    print("  ERR", d); sys.exit()
rows=[]
for s in d["data"]["result"]:
    rows.append((" ".join(s["metric"].get(k,"-") for k in keep), float(s["value"][1])))
for name,v in sorted(rows):
    print("  %-60s %.4f" % (name, v))
'
done

# Print the node's disk, image and memory figures from the kubelet's /stats/summary, read from stdin.
import json, sys
d = json.load(sys.stdin)
n = d["node"]
G = 1e9
fs = n["fs"]
im = n["runtime"]["imageFs"]
print("node fs: capacity %.2f GB, used %.2f GB, available %.2f GB, inodes used %s of %s" % (
    fs["capacityBytes"] / G, fs["usedBytes"] / G, fs["availableBytes"] / G, fs.get("inodesUsed"), fs.get("inodes")))
print("image fs: capacity %.2f GB, used %.2f GB" % (im["capacityBytes"] / G, im["usedBytes"] / G))
print("node memory: working set %.2f GiB, rss %.2f GiB, available %.2f GiB" % (
    n["memory"]["workingSetBytes"] / 2**30, n["memory"]["rssBytes"] / 2**30, n["memory"]["availableBytes"] / 2**30))
total = sum(p.get("ephemeral-storage", {}).get("usedBytes", 0) for p in d["pods"])
print("ephemeral storage used by all %d pods: %.2f GB" % (len(d["pods"]), total / G))
