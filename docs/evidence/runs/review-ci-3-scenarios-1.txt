fake_kustomize_fails() { # A kustomize that fails once told to: a render that breaks.
  rm -f "${CLONE:?}/.tools/bin/kustomize"
  cat >"$CLONE/.tools/bin/kustomize" <<SH
#!/usr/bin/env bash
if [[ -e "\$STUB/kustomize-fails" && "\$1" == build ]]; then echo "stub kustomize: failing as told" >&2; exit 1; fi
exec "$ROOT/.tools/bin/kustomize" "\$@"
SH
  chmod +x "$CLONE/.tools/bin/kustomize"
}
real_kustomize() { rm -f "${CLONE:?}/.tools/bin/kustomize" "${STUB:?}/kustomize-fails"; ln -s "$ROOT/.tools/bin/kustomize" "$CLONE/.tools/bin/kustomize"; }

# 1. A conflict in the overlay, with someone else's work staged in this checkout.
rv_conflict_with_staged_work() {
  local status=0
  RV_NAME=conflict-staged
  promote_prod_sim_committed
  set_image prod-sim artifacts.plane.so/makeplane/plane-frontend "docker.io/fixture/plane-frontend:v0@$DIGEST_B"
  commit_a_change_to_helm 'a digest changed by hand, so not a promotion'
  echo "someone's new file" >"$CLONE/docs/staged-by-someone.txt"
  echo "someone's edit" >>"$CLONE/README.md"
  git -C "$CLONE" add docs/staged-by-someone.txt README.md
  obs "before: staged: $(git -C "$CLONE" diff --cached --name-only | tr '\n' ' ')"
  rollback prod-sim || status=$?
  snapshot "$status"
  obs "after: staged: $(git -C "$CLONE" diff --cached --name-only | tr '\n' ' ')"
  obs "after: new file on disk: $([ -e "$CLONE/docs/staged-by-someone.txt" ] && echo yes || echo NO)"
  obs "after: the edit in README.md on disk: $(grep -c "someone's edit" "$CLONE/README.md")"
  expect "a refusal" [ "$status" -ne 0 ]
  expect "the staged new file survives on disk" [ -e "$CLONE/docs/staged-by-someone.txt" ]
  expect "the staged edit survives on disk" grep -q "someone's edit" "$CLONE/README.md"
  git -C "$CLONE" reset --quiet --hard HEAD; rm -f "${CLONE:?}/docs/staged-by-someone.txt"
}

# 4. git commit fails after the revert is staged. rollback.sh:147 says a failure here leaves nothing behind.
rv_commit_fails_after_revert() {
  local status=0
  RV_NAME=commit-fails
  promote_prod_sim_committed
  printf '#!/bin/sh\necho "hook: no" >&2\nexit 1\n' >"$CLONE/.git/hooks/pre-commit"
  chmod +x "$CLONE/.git/hooks/pre-commit"
  rollback prod-sim || status=$?
  snapshot "$status"
  rm -f "${CLONE:?}/.git/hooks/pre-commit"
  expect "a refusal" [ "$status" -ne 0 ]
  expect "nothing left in progress" [ ! -e "$CLONE/.git/REVERT_HEAD" ]
  expect "helm and gitops clean" equal "$(git -C "$CLONE" status --porcelain -- helm gitops)" ""
  status=0
  rollback prod-sim || status=$?
  obs "a second run, the hook gone:"; snapshot "$status" 4
  expect "a second run finishes the rollback" [ "$status" -eq 0 ]
  git -C "$CLONE" revert --abort 2>/dev/null || true; git -C "$CLONE" reset --quiet --hard HEAD
}

lock_after_render() { # A kustomize that leaves .git/index.lock behind once told to: another process's git add, held through the next few commands.
  rm -f "${CLONE:?}/.tools/bin/kustomize"
  cat >"$CLONE/.tools/bin/kustomize" <<SH
#!/usr/bin/env bash
"$ROOT/.tools/bin/kustomize" "\$@"
status=\$?
if [[ -e "\$STUB/lock-now" && "\$1" == build ]]; then touch "$CLONE/.git/index.lock"; fi
exit \$status
SH
  chmod +x "$CLONE/.tools/bin/kustomize"
}
# A concurrent git add holds the index while rollback goes to stage its revert.
rv2_rollback_index_lock() {

  RV_NAME=rollback-index-lock

  real_kustomize
  snapshot "$status" 5
  expect "a refusal" [ "$status" -ne 0 ]
  expect "nothing left in progress" [ ! -e "$CLONE/.git/REVERT_HEAD" ]
  expect "helm and gitops clean" equal "$(git -C "$CLONE" status --porcelain -- helm gitops)" ""
  status=0
  rollback prod-sim || status=$?
  obs "a second run, the lock gone:"; snapshot "$status" 4
  expect "a second run finishes the rollback" [ "$status" -eq 0 ]
  git -C "$CLONE" revert --abort 2>/dev/null || true; git -C "$CLONE" reset --quiet --hard HEAD
}
# The same concurrent git add during a promotion, after its dump. The report says the script restores the overlay.
rv2_promote_index_lock() {

  RV_NAME=promote-index-lock

  lock_after_render
  touch "$STUB/lock-now"

  real_kustomize

  in_clone env SKIP_REGISTRY_CHECK=yes CONFIRM=yes ./scripts/promote.sh prod-sim || status=$?
  obs "a second run, the lock gone:"; snapshot "$status" 3
  git -C "$CLONE" reset --quiet --hard HEAD
}
