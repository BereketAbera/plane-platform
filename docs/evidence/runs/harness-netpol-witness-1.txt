== docker-fails-later/docker
#!/bin/bash
# Lets the first witness through, the sentinel's, and fails every later one. Everything else goes to the real docker.
count="<scratch>/docker-fails-later/count"
if [[ "$1 $2" == "exec -i" ]]; then
  n=$(cat "$count" 2>/dev/null || echo 0)
  n=$((n + 1))
  echo "$n" >"$count"
  if [[ "$n" -gt 1 ]]; then
    echo "stub docker: the daemon went away after the sentinel" >&2
    exit 1
  fi
fi
exec "/usr/local/bin/docker" "$@"
