--- sim.py
# The tick model the 11m reads from, as arithmetic and not a measurement: an informer resync every 120s, a jitter in [0, 60) per
# Application per tick, a look only if more than 120s have passed since reconciledAt, a failed look that leaves reconciledAt alone,
# the first failure starting a 180s grace, and a ref cache shared by every Application that lives 120s.
import random
def run(apps=14):
    c=random.uniform(1200,1320)   # commit time
    r={a:random.uniform(1000,1100) for a in range(apps)}
    fetched=c-random.uniform(0,120); old=True
    first=None
    k=0
    while True:
        k+=1
        t0=120*k
        looks=sorted((t0+random.uniform(0,60),a) for a in range(apps))
        for t,a in looks:
            if t-r[a]<=120: continue
            if t-fetched>=120:
                fetched=t
            sees_new = fetched>=c
            if not sees_new:
                r[a]=t
                continue
            if a!=0:
                r[a]=t
                continue
            if first is None:
                first=t
            elif t-first>180:
                return t-c
random.seed(1)
v=sorted(run() for _ in range(100000))
n=len(v)
print("seconds from commit to Unknown, 100000 runs: min %.0f median %.0f p95 %.0f p99.9 %.0f max %.0f"%(v[0],v[n//2],v[int(n*.95)],v[int(n*.999)],v[-1]))
print("share at or under the one real sample, 357s: %.0f%%"%(100*sum(1 for x in v if x<=357)/n))
