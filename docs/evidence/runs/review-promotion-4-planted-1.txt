=== plant.py
#!/usr/bin/env python3
# plant.py <file> <old> <new>: replaces exactly one occurrence, or fails.
import sys
p, old, new = sys.argv[1:4]
s = open(p).read()
n = s.count(old)
if n != 1:
    sys.exit("matches: %d for %r" % (n, old[:60]))
open(p, "w").write(s.replace(old, new))
