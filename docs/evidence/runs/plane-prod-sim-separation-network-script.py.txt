#!/usr/bin/env python3
"""Real connection attempts between plane-dev and plane-prod-sim, from real running pods.

Reads pods, Services and pod IPs from the cluster. Prints no credential: the only values that leave a
pod are verdicts, timings, addresses and an error class, and the whole file is scanned afterwards.

For each direction:
  1. Every source pod that has a shell or an interpreter dials the other environment's postgres,
     valkey, rabbitmq (AMQP and management) and silo (S3 and console), by Service name and by pod IP.
  2. The same pod dials its own environment's data tier in the same batch, so a dead network stack
     or a broken probe cannot pass as isolation.
  3. A pod in the target environment dials the same targets in the same moment, so a target that is
     down cannot pass as isolation.
  4. The real clients the app uses (psycopg, redis-py, amqp, boto3), with the source environment's
     own credentials, dial their own and the other environment's servers.

A deny counts only as a TIMEOUT: refused means the packet arrived, and a DNS failure says nothing.
"""
from __future__ import annotations

import concurrent.futures
import datetime as dt
import json
import subprocess
import sys

ROOT = "/Users/bek/dev/pr/plane-platform"
KUBECTL = f"{ROOT}/.tools/bin/kubectl"
KUBECONFIG = f"{ROOT}/.local/kubeconfig"
TIMEOUT = 5
ENVS = {"dev": "plane-dev", "prod-sim": "plane-prod-sim"}
TARGETS = [("postgres", "plane-pgdb", 5432), ("valkey", "plane-redis", 6379), ("rabbitmq-amqp", "plane-rabbitmq", 5672),
           ("rabbitmq-mgmt", "plane-rabbitmq", 15672), ("silo-s3", "plane-minio", 9000), ("silo-console", "plane-minio", 9090)]
POD_OF_SERVICE = {"plane-pgdb": "plane-pgdb-wl-0", "plane-redis": "plane-redis-wl-0", "plane-rabbitmq": "plane-rabbitmq-wl-0",
                  "plane-minio": "plane-minio-wl-0"}
# Ports the app tier may use in its own environment, per the app-tier-to-data policy.
OWN_ALLOWED = {"postgres", "valkey", "rabbitmq-amqp", "silo-s3"}

PY_PROBE = r"""
import concurrent.futures, json, socket, sys, time
targets = json.load(sys.stdin)
def one(t):
    started = time.monotonic()
    try:
        ip = socket.getaddrinfo(t["host"], t["port"], socket.AF_INET, socket.SOCK_STREAM)[0][4][0]
    except OSError as e:
        return {**t, "ip": None, "verdict": "DNS-FAILED", "seconds": round(time.monotonic() - started, 2), "detail": type(e).__name__}
    s = socket.socket()
    s.settimeout(%(timeout)d)
    started = time.monotonic()
    try:
        s.connect((ip, t["port"]))
        verdict = "CONNECTED"
    except TimeoutError:
        verdict = "TIMEOUT"
    except ConnectionRefusedError:
        verdict = "REFUSED"
    except OSError as e:
        verdict = "ERROR:" + type(e).__name__
    finally:
        s.close()
    return {**t, "ip": ip, "verdict": verdict, "seconds": round(time.monotonic() - started, 2)}
with concurrent.futures.ThreadPoolExecutor(32) as pool:
    print(json.dumps(list(pool.map(one, targets))))
""" % {"timeout": TIMEOUT}

NODE_PROBE = r"""
const net = require("net"), dns = require("dns");
let raw = ""; process.stdin.on("data", d => raw += d); process.stdin.on("end", async () => {
  const targets = JSON.parse(raw);
  const one = t => new Promise(resolve => {
    dns.lookup(t.host, { family: 4 }, (err, ip) => {
      if (err) return resolve({ ...t, ip: null, verdict: "DNS-FAILED", seconds: 0 });
      const started = Date.now();
      const s = net.connect({ host: ip, port: t.port });
      const done = v => { s.destroy(); resolve({ ...t, ip, verdict: v, seconds: (Date.now() - started) / 1000 }); };
      s.setTimeout(%(ms)d, () => done("TIMEOUT"));
      s.on("connect", () => done("CONNECTED"));
      s.on("error", e => done(e.code === "ECONNREFUSED" ? "REFUSED" : "ERROR:" + e.code));
    });
  });
  console.log(JSON.stringify(await Promise.all(targets.map(one))));
});
""" % {"ms": TIMEOUT * 1000}

# busybox nc. -w bounds the connect, and the elapsed time and message separate timeout from refused.
SH_PROBE = r"""
tmp=$(mktemp -d)
i=0
while read -r name host port via; do
  [ -n "$name" ] || continue
  i=$((i+1))
  (
    started=$(date +%%s)
    if out=$(nc -w %(timeout)d "$host" "$port" </dev/null 2>&1); then rc=0; else rc=$?; fi
    ended=$(date +%%s)
    printf '%%s %%s %%s %%s %%s %%s %%s\n' "$name" "$host" "$port" "$via" "$rc" "$((ended-started))" "$(printf '%%s' "$out" | tr ' \n' '__' | cut -c1-80)" >"$tmp/$i"
  ) &
done
wait
cat "$tmp"/*
rm -rf "$tmp"
""" % {"timeout": TIMEOUT}

# The clients the app itself uses, with this pod's own credentials from its environment. argv[1] is the target namespace.
REAL_CLIENTS = r"""
import json, os, sys, time, urllib.parse
ns = sys.argv[1]
host = lambda service: f"{service}.{ns}.svc.cluster.local"
def attempt(name, fn):
    started = time.monotonic()
    try:
        detail = fn()
        verdict = "CONNECTED"
    except BaseException as exc:
        verdict = type(exc).__name__
        detail = str(exc).replace("\n", " ")[:160]
    print(json.dumps({"client": name, "target_namespace": ns, "verdict": verdict, "seconds": round(time.monotonic() - started, 2), "detail": detail}))
def pg():
    import psycopg
    u = urllib.parse.urlsplit(os.environ["DATABASE_URL"])
    with psycopg.connect(host=host("plane-pgdb"), dbname=u.path.lstrip("/"), user=u.username, password=u.password, connect_timeout=%(timeout)d) as c:
        return "SELECT 1 -> %%s" %% c.execute("select 1").fetchone()[0]
def valkey():
    import redis
    return "PING -> %%s" %% redis.Redis(host=host("plane-redis"), port=6379, socket_connect_timeout=%(timeout)d, socket_timeout=%(timeout)d).ping()
def amqp_():
    import amqp
    u = urllib.parse.urlsplit(os.environ["AMQP_URL"])
    c = amqp.Connection(host=host("plane-rabbitmq") + ":5672", userid=urllib.parse.unquote(u.username), password=urllib.parse.unquote(u.password), connect_timeout=%(timeout)d)
    c.connect()
    server = c.server_properties.get("product")
    c.close()
    return "AMQP open, server %%s" %% server
def silo():
    import boto3
    from botocore.config import Config
    s3 = boto3.client("s3", endpoint_url=f"http://{host('plane-minio')}:9000", aws_access_key_id=os.environ["AWS_ACCESS_KEY_ID"],
                      aws_secret_access_key=os.environ["AWS_SECRET_ACCESS_KEY"], region_name="us-east-1",
                      config=Config(signature_version="s3v4", connect_timeout=%(timeout)d, read_timeout=%(timeout)d, retries={"total_max_attempts": 1}))
    return "ListBuckets -> %%s" %% [b["Name"] for b in s3.list_buckets()["Buckets"]]
attempt("psycopg", pg)
attempt("redis-py", valkey)
attempt("amqp", amqp_)
attempt("boto3", silo)
""" % {"timeout": TIMEOUT}


def now() -> str:
    return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def kubectl(*args: str, stdin: str | None = None, timeout: float = 90) -> subprocess.CompletedProcess:
    return subprocess.run([KUBECTL, "--kubeconfig", KUBECONFIG, "--context", "kind-plane-platform", *args], input=stdin,
                          capture_output=True, text=True, timeout=timeout, check=False)


def pods(namespace: str) -> dict:
    listed = json.loads(kubectl("-n", namespace, "get", "pods", "-o", "json").stdout)["items"]
    return {p["metadata"]["name"]: p for p in listed if p["status"]["phase"] == "Running" and p["metadata"].get("deletionTimestamp") is None}


def named(all_pods: dict, prefix: str) -> str:
    matches = [name for name in all_pods if name.startswith(prefix)]
    if not matches:
        raise SystemExit(f"no running pod {prefix}")
    return matches[0]


def targets_for(namespace: str, pod_ips: dict) -> list[dict]:
    out = []
    for label, service, port in TARGETS:
        out.append({"name": label, "host": f"{service}.{namespace}.svc.cluster.local", "port": port, "via": "service-dns"})
        out.append({"name": label, "host": pod_ips[POD_OF_SERVICE[service]], "port": port, "via": "pod-ip"})
    return out


def probe(namespace: str, pod: str, container: str, kind: str, targets: list[dict]) -> list[dict]:
    if kind == "python":
        done = kubectl("-n", namespace, "exec", "-i", pod, "-c", container, "--", "python", "-c", PY_PROBE, stdin=json.dumps(targets))
    elif kind == "node":
        done = kubectl("-n", namespace, "exec", "-i", pod, "-c", container, "--", "node", "-e", NODE_PROBE, stdin=json.dumps(targets))
    else:
        lines = "\n".join(f"{t['name']} {t['host']} {t['port']} {t['via']}" for t in targets) + "\n"
        done = kubectl("-n", namespace, "exec", "-i", pod, "-c", container, "--", "sh", "-c", SH_PROBE, stdin=lines)
    if done.returncode != 0:
        return [{**t, "ip": None, "verdict": f"EXEC-FAILED rc={done.returncode}", "seconds": 0, "detail": done.stderr.strip()[:120]} for t in targets]
    if kind != "shell":
        return json.loads(done.stdout)
    results = []
    for line in done.stdout.splitlines():
        name, host, port, via, rc, seconds, message = (line.split(" ", 6) + [""])[:7]
        text = message.lower()
        if rc == "0":
            verdict = "CONNECTED"
        elif "refused" in text:
            verdict = "REFUSED"
        elif "bad_address" in text:
            verdict = "DNS-FAILED"
        elif int(seconds) >= TIMEOUT - 1:
            verdict = "TIMEOUT"
        else:
            verdict = f"ERROR:{message}"
        results.append({"name": name, "host": host, "port": int(port), "via": via, "verdict": verdict, "seconds": int(seconds), "ip": host if via == "pod-ip" else "-"})
    return results


def row(moment: str, direction: str, source_ns: str, source: str, source_ip: str, host_network: bool, result: dict) -> str:
    return "\t".join(str(v) for v in (moment, direction, f"{source_ns}/{source}", source_ip, f"hostNetwork={str(host_network).lower()}",
                                      result["name"], result["host"], result.get("ip", ""), result["port"], result["via"], result["verdict"], result["seconds"]))


def direction(source_env: str, target_env: str) -> tuple[int, int]:
    source_ns, target_ns = ENVS[source_env], ENVS[target_env]
    failures = passes = 0
    source_pods, target_pods = pods(source_ns), pods(target_ns)
    target_ips = {name: p["status"]["podIP"] for name, p in target_pods.items()}
    source_ips = {name: p["status"]["podIP"] for name, p in source_pods.items()}
    other_targets = targets_for(target_ns, target_ips)
    own_targets = targets_for(source_ns, source_ips)
    sources = [("plane-api-wl", "plane-api", "python"), ("plane-worker-wl", "plane-worker", "python"),
               ("plane-beat-worker-wl", "plane-beat-worker", "python"), ("plane-live-wl", "plane-live", "node"),
               ("plane-space-wl", "plane-space", "node"), ("plane-web-wl", "plane-web", "shell"), ("plane-admin-wl", "plane-admin", "shell"),
               ("plane-pgdb-wl-0", "plane-pgdb", "shell"), ("plane-redis-wl-0", "plane-redis", "shell"), ("plane-rabbitmq-wl-0", "plane-rabbitmq", "shell")]
    label = f"{source_env}->{target_env}"
    print(f"\n## {label}: every source pod dials the other environment. A pass is a TIMEOUT and nothing else. {now()}")
    print("utc\tdirection\tsource\tsource_ip\thostNetwork\ttarget\thost\tip\tport\tvia\tverdict\tseconds")

    def run_source(entry):
        prefix, container, kind = entry
        pod = named(source_pods, prefix)
        moment = now()
        return pod, moment, probe(source_ns, pod, container, kind, other_targets), source_pods[pod]

    with concurrent.futures.ThreadPoolExecutor(len(sources)) as pool:
        batches = list(pool.map(run_source, sources))
    for pod, moment, results, spec in batches:
        for result in results:
            print(row(moment, label, source_ns, pod, source_ips[pod], bool(spec["spec"].get("hostNetwork")), result))
            if result["verdict"] == "TIMEOUT":
                passes += 1
            else:
                failures += 1
                print(f"FAIL: {pod} to {result['host']}:{result['port']} read {result['verdict']}, want TIMEOUT")

    # Same moment: the targets answer a pod in their own environment.
    print(f"\n## {label}: reference, a pod inside {target_ns} dials the same targets in the same moment. {now()}")
    reference = named(target_pods, "plane-api-wl")
    for result in probe(target_ns, reference, "plane-api", "python", other_targets):
        print(row(now(), f"{target_env}->{target_env}", target_ns, reference, target_ips[reference], bool(target_pods[reference]["spec"].get("hostNetwork")), result))
        want = "CONNECTED" if result["name"] in OWN_ALLOWED else "TIMEOUT"
        if result["verdict"] == want:
            passes += 1
        else:
            failures += 1
            print(f"FAIL: reference to {result['host']}:{result['port']} read {result['verdict']}, want {want}")

    print(f"\n## {label}: reference, the source pods with a data-tier rule dial their own environment. {now()}")
    for prefix, container, kind in sources[:3]:
        pod = named(source_pods, prefix)
        for result in probe(source_ns, pod, container, kind, own_targets):
            print(row(now(), f"{source_env}->{source_env}", source_ns, pod, source_ips[pod], bool(source_pods[pod]["spec"].get("hostNetwork")), result))
            want = "CONNECTED" if result["name"] in OWN_ALLOWED else "TIMEOUT"
            if result["verdict"] == want:
                passes += 1
            else:
                failures += 1
                print(f"FAIL: {pod} to its own {result['host']}:{result['port']} read {result['verdict']}, want {want}")

    print(f"\n## {label}: the real clients, with {source_ns}'s own credentials from the pod's environment. {now()}")
    api = named(source_pods, "plane-api-wl")
    for namespace, want_connected in ((source_ns, True), (target_ns, False), (source_ns, True)):
        done = kubectl("-n", source_ns, "exec", api, "-c", "plane-api", "--", "python", "-c", REAL_CLIENTS, namespace)
        if done.returncode != 0:
            failures += 1
            print(f"FAIL: real clients in {api} exited {done.returncode}: {done.stderr.strip()[:200]}")
            continue
        for line in done.stdout.splitlines():
            record = json.loads(line)
            print(json.dumps({"utc": now(), "from": f"{source_ns}/{api}", **record}))
            timed_out = record["verdict"] in ("ConnectionTimeout", "ConnectTimeoutError", "TimeoutError", "timeout", "ConnectTimeoutError", "OperationalError") \
                and ("timeout" in record["detail"].lower() or "timed out" in record["detail"].lower() or record["seconds"] >= TIMEOUT - 1)
            if want_connected and record["verdict"] == "CONNECTED":
                passes += 1
            elif not want_connected and timed_out:
                passes += 1
            else:
                failures += 1
                print(f"FAIL: {record['client']} to {namespace}: {record['verdict']}, wanted {'a connection' if want_connected else 'a timeout'}")

    # The two binaries the data pods ship, so the attempt is a real protocol client there too.
    print(f"\n## {label}: psql and valkey-cli inside {source_ns}'s data pods. {now()}")
    target_pg = f"plane-pgdb.{target_ns}.svc.cluster.local"
    target_valkey = f"plane-redis.{target_ns}.svc.cluster.local"
    command = f'started=$(date +%s); pg_isready -h {target_pg} -t {TIMEOUT}; rc=$?; echo "pg_isready rc=$rc seconds=$(( $(date +%s)-started ))"'
    done = kubectl("-n", source_ns, "exec", "plane-pgdb-wl-0", "-c", "plane-pgdb", "--", "sh", "-c", command)
    print(f"{now()} {source_ns}/plane-pgdb-wl-0 -> {target_pg}: {done.stdout.strip()} {done.stderr.strip()[:100]}")
    if "rc=2" in done.stdout:
        passes += 1
    else:
        failures += 1
        print("FAIL: pg_isready should report no response (rc=2)")
    command = f'started=$(date +%s); timeout {TIMEOUT} valkey-cli -h {target_valkey} ping; rc=$?; echo "valkey-cli rc=$rc seconds=$(( $(date +%s)-started ))"'
    done = kubectl("-n", source_ns, "exec", "plane-redis-wl-0", "-c", "plane-redis", "--", "sh", "-c", command)
    print(f"{now()} {source_ns}/plane-redis-wl-0 -> {target_valkey}: {done.stdout.strip()} {done.stderr.strip()[:100]}")
    if "PONG" not in done.stdout and "seconds=" in done.stdout and int(done.stdout.rsplit("seconds=", 1)[1].split()[0]) >= TIMEOUT - 1:
        passes += 1
    else:
        failures += 1
        print("FAIL: valkey-cli should have hung until the timeout")
    return passes, failures


def main() -> int:
    print(f"# plane-prod-sim separation, network. Real attempts from real pods. {now()}")
    print("# command: python3 -B sep_network.py (a copy of the script is saved beside this file)")
    total_pass = total_fail = 0
    for source, target in (("dev", "prod-sim"), ("prod-sim", "dev")):
        passed, failed = direction(source, target)
        total_pass += passed
        total_fail += failed
    print(f"\nRESULT checks passed={total_pass} failed={total_fail} {now()}")
    return 1 if total_fail else 0


if __name__ == "__main__":
    sys.exit(main())
